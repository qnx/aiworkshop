cmake_minimum_required(VERSION 3.16)
project(handtrack C CXX)

set(CMAKE_CXX_STANDARD 20)   # std::format in the status line
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)        # Header, ColorRGBA in HandSet
find_package(geometry_msgs REQUIRED)   # Point for the landmarks
find_package(ament_index_cpp REQUIRED) # locate the installed models at runtime
find_package(rosidl_default_generators REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/Hand.msg"
  "msg/HandSet.msg"
  DEPENDENCIES std_msgs geometry_msgs
)


# Fetch the ML models if they aren't already downloaded.
set(MODEL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/models)
set(MP_ASSETS https://storage.googleapis.com/mediapipe-assets)
foreach(m palm_detection_lite hand_landmark_lite)
    if(NOT EXISTS ${MODEL_DIR}/${m}.tflite)
        message(STATUS "Downloading ${m}.tflite")
        file(DOWNLOAD ${MP_ASSETS}/${m}.tflite ${MODEL_DIR}/${m}.tflite SHOW_PROGRESS STATUS dl)
        list(GET dl 0 rc)
        if(NOT rc EQUAL 0)
            file(REMOVE ${MODEL_DIR}/${m}.tflite)
            message(FATAL_ERROR "download of ${m}.tflite failed: ${dl}")
        endif()
    endif()
endforeach()

find_package(Threads REQUIRED)
find_package(OpenCV REQUIRED)
find_library(TFLITE_LIB tensorflow-lite REQUIRED)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)

# todo workaround for flatbuffers being the wrong version
set(FLATBUFFERS_SHIM -include ${CMAKE_CURRENT_SOURCE_DIR}/flatbuffers_version_shim.hpp)

add_executable(handtrack_node handtrack_node.cpp handtracking.cpp rpi_cam.cpp handtrack_gui.cpp)
target_compile_options(handtrack_node PRIVATE ${FLATBUFFERS_SHIM})
target_link_libraries(handtrack_node
    ${TFLITE_LIB}
    ${OpenCV_LIBS}
    camapi              # RpiCam viewfinder capture
    PkgConfig::GTK4     # HandtrackGui output window
    Threads::Threads)

ament_target_dependencies(handtrack_node
  "rclcpp"
  "std_msgs"
  "geometry_msgs"
  "ament_index_cpp"
)

# The messages are generated by this same package, so they cannot be picked up
# with find_package -- link the generated typesupport directly.
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")
target_link_libraries(handtrack_node "${cpp_typesupport_target}")

ament_export_dependencies(rosidl_default_runtime)

# The node resolves models out of the share directory, not its working directory.
install(DIRECTORY models DESTINATION share/${PROJECT_NAME})

install(TARGETS
  handtrack_node
  DESTINATION lib/${PROJECT_NAME}
)

ament_package()
